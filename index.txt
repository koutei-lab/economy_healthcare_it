import React, { useState, useEffect, useRef } from 'react';
import { initializeApp } from 'firebase/app';
import { getAuth, signInAnonymously, signInWithCustomToken, onAuthStateChanged } from 'firebase/auth';
import { getFirestore, doc, collection, onSnapshot, setDoc, updateDoc, addDoc, deleteDoc } from 'firebase/firestore';
import { Newspaper, Bookmark, Settings, CheckCircle, Circle, Copy, Compass, RefreshCw, Sun, Loader2 } from 'lucide-react';

// === Firebase Initialization ===
const firebaseConfig = JSON.parse(__firebase_config);
const app = initializeApp(firebaseConfig);
const auth = getAuth(app);
const db = getFirestore(app);
const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';

// === Utility Functions ===
const getTodayString = () => {
    const d = new Date();
    return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
};

const isPastFetchTime = (timeStr) => {
    if (!timeStr) return false;
    const now = new Date();
    const [h, m] = timeStr.split(':').map(Number);
    const target = new Date(now.getFullYear(), now.getMonth(), now.getDate(), h, m);
    return now >= target;
};

// === UI Components ===
const MorningNewsIcon = ({ className = "w-8 h-8" }) => (
    <svg viewBox="0 0 100 100" className={className} xmlns="http://www.w3.org/2000/svg">
        <defs>
            <linearGradient id="grad" x1="0%" y1="0%" x2="100%" y2="100%">
                <stop offset="0%" stopColor="#FDE68A" />
                <stop offset="100%" stopColor="#F97316" />
            </linearGradient>
        </defs>
        <rect width="100" height="100" rx="25" fill="url(#grad)" />
        <path d="M25 35h50v40H25z" fill="#FFF" opacity="0.9" rx="4" />
        <path d="M32 45h36M32 55h36M32 65h20" stroke="#F97316" strokeWidth="6" strokeLinecap="round" />
        <circle cx="70" cy="25" r="10" fill="#FFF" opacity="0.9" />
    </svg>
);

const NavItem = ({ icon, label, active, onClick }) => (
    <button onClick={onClick} className={`flex flex-col items-center gap-1 w-20 transition-colors ${active ? 'text-orange-500' : 'text-gray-400 hover:text-gray-600'}`}>
        {React.cloneElement(icon, { size: 24, strokeWidth: active ? 2.5 : 2 })}
        <span className="text-[10px] font-bold">{label}</span>
    </button>
);

// === Main App Component ===
export default function App() {
    const [user, setUser] = useState(null);
    const [allArticles, setAllArticles] = useState([]);
    const [settings, setSettings] = useState({ time: "07:00", lastFetchedDate: "", readHistory: [] });
    const [settingsLoaded, setSettingsLoaded] = useState(false);
    
    const [tab, setTab] = useState('home');
    const [categoryFilter, setCategoryFilter] = useState('All');
    const [isFetching, setIsFetching] = useState(false);
    const [toast, setToast] = useState(null);
    const autoFetchAttempted = useRef(false);

    const showToast = (msg) => {
        setToast(msg);
        setTimeout(() => setToast(null), 3000);
    };

    // 1. Authentication
    useEffect(() => {
        const initAuth = async () => {
            if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                await signInWithCustomToken(auth, __initial_auth_token);
            } else {
                await signInAnonymously(auth);
            }
        };
        initAuth();
        const unsubscribe = onAuthStateChanged(auth, setUser);
        return () => unsubscribe();
    }, []);

    // 2. Data Subscriptions
    useEffect(() => {
        if (!user) return;

        // Sub articles
        const articlesRef = collection(db, 'artifacts', appId, 'users', user.uid, 'articles');
        const unsubArticles = onSnapshot(articlesRef, (snapshot) => {
            const arr = [];
            snapshot.forEach(d => arr.push({ id: d.id, ...d.data() }));
            arr.sort((a, b) => b.createdAt - a.createdAt);
            setAllArticles(arr);
        }, (err) => console.error("Articles Error:", err));

        // Sub settings
        const settingsRef = doc(db, 'artifacts', appId, 'users', user.uid, 'config', 'settings');
        const unsubSettings = onSnapshot(settingsRef, (docSnap) => {
            if (docSnap.exists()) {
                setSettings(prev => ({ ...prev, ...docSnap.data() }));
            }
            setSettingsLoaded(true);
        }, (err) => console.error("Settings Error:", err));

        return () => {
            unsubArticles();
            unsubSettings();
        };
    }, [user]);

    // 3. Auto Fetch Trigger
    useEffect(() => {
        if (!user || !settingsLoaded || autoFetchAttempted.current || isFetching) return;

        const todayStr = getTodayString();
        if (settings.time && isPastFetchTime(settings.time) && settings.lastFetchedDate !== todayStr) {
            autoFetchAttempted.current = true;
            handleFetchNews();
        }
    }, [user, settingsLoaded, settings.time, settings.lastFetchedDate, isFetching]);

    // === Core Business Logic ===
    const handleFetchNews = async () => {
        if (isFetching || !user) return;
        setIsFetching(true);
        try {
            const historyContext = settings.readHistory.slice(-50).join(', ');
            const prompt = `あなたは優秀なエンジニア兼キュレーターです。経済、医療、ITの3つの分野の過去24時間以内の最新ニュースを検索し、提供してください。
            
【厳守する条件】
1. 経済、医療、ITの各分野から3つずつ、合計9つのニュースを取得してください。
2. summaryは300〜400文字程度で、分かりやすく詳細にプレーンテキストで要約してください。
3. 以下のタイトルに類似する既読済みのニュースは**絶対に含めない**でください。
既読リスト: [${historyContext}]
4. categoryフィールドは必ず "経済", "医療", "IT" のいずれかを指定してください。
5. 引用元のメディア名(sourceName)と実在するURL(sourceUrl)を正確に記載してください。`;

            const payload = {
                contents: [{ parts: [{ text: prompt }] }],
                tools: [{ google_search: {} }],
                generationConfig: {
                    responseMimeType: "application/json",
                    responseSchema: {
                        type: "ARRAY",
                        items: {
                            type: "OBJECT",
                            properties: {
                                title: { type: "STRING" },
                                summary: { type: "STRING" },
                                sourceName: { type: "STRING" },
                                sourceUrl: { type: "STRING" },
                                category: { type: "STRING" }
                            },
                            propertyOrdering: ["title", "summary", "sourceName", "sourceUrl", "category"]
                        }
                    }
                }
            };

            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=`;
            const res = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });

            if (!res.ok) throw new Error("API Request Failed");
            
            const data = await res.json();
            const text = data.candidates?.[0]?.content?.parts?.[0]?.text;
            
            if (text) {
                const parsed = JSON.parse(text);
                const todayStr = getTodayString();
                
                // 古い「未保管」の記事を削除して画面をクリーンにする
                const oldUnsaved = allArticles.filter(a => !a.isSaved);
                for (const a of oldUnsaved) {
                    await deleteDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'articles', a.id));
                }

                // 新しい記事を追加
                for (const item of parsed) {
                    await addDoc(collection(db, 'artifacts', appId, 'users', user.uid, 'articles'), {
                        ...item,
                        isRead: false,
                        isSaved: false,
                        fetchBatchId: todayStr,
                        createdAt: Date.now()
                    });
                }

                // 最終取得日を更新
                await setDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'config', 'settings'), {
                    lastFetchedDate: todayStr
                }, { merge: true });

                showToast('今日の最新ニュースを取得しました ✨');
                setTab('home');
            } else {
                throw new Error("Invalid Response Format");
            }
        } catch (e) {
            console.error("Fetch News Error:", e);
            showToast('ニュースの取得に失敗しました。再試行してください。');
        } finally {
            setIsFetching(false);
        }
    };

    const toggleRead = async (article) => {
        if (!user) return;
        const newIsRead = !article.isRead;
        const articleRef = doc(db, 'artifacts', appId, 'users', user.uid, 'articles', article.id);
        
        await updateDoc(articleRef, { isRead: newIsRead });
        
        if (newIsRead) {
            // 既読リストに追加（重複排除と最大100件の維持）
            const newHistory = [...new Set([...settings.readHistory, article.title])].slice(-100);
            await setDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'config', 'settings'), {
                readHistory: newHistory
            }, { merge: true });
        }
    };

    const toggleSave = async (article) => {
        if (!user) return;
        const articleRef = doc(db, 'artifacts', appId, 'users', user.uid, 'articles', article.id);
        await updateDoc(articleRef, { isSaved: !article.isSaved });
        showToast(!article.isSaved ? '記事を保管しました' : '保管を解除しました');
    };

    const handleCopyUrl = async (url) => {
        try {
            await navigator.clipboard.writeText(url);
            showToast('URLをコピーしました！');
        } catch (e) {
            showToast('コピーに失敗しました');
        }
    };

    const updateTimeSetting = async (e) => {
        const newTime = e.target.value;
        if (!user) return;
        await setDoc(doc(db, 'artifacts', appId, 'users', user.uid, 'config', 'settings'), {
            time: newTime
        }, { merge: true });
        showToast('更新時間を保存しました');
    };

    // === Render Helpers ===
    const getCategoryStyles = (category) => {
        if (category?.includes('経済')) return "bg-blue-100 text-blue-800 border-blue-200";
        if (category?.includes('医療')) return "bg-rose-100 text-rose-800 border-rose-200";
        if (category?.includes('IT')) return "bg-emerald-100 text-emerald-800 border-emerald-200";
        return "bg-gray-100 text-gray-800 border-gray-200";
    };

    const renderArticleCard = (article) => (
        <div key={article.id} className={`bg-white rounded-2xl p-5 shadow-sm border border-orange-100/50 flex flex-col gap-4 transition-all duration-300 ${article.isRead ? 'opacity-60 scale-[0.98]' : 'hover:shadow-md'}`}>
            <div className="flex justify-between items-start gap-3">
                <h3 className="font-bold text-[17px] text-gray-800 leading-snug flex-1">{article.title}</h3>
                <span className={`text-[11px] font-bold px-2.5 py-1 rounded-full whitespace-nowrap border ${getCategoryStyles(article.category)}`}>
                    {article.category || 'ニュース'}
                </span>
            </div>
            
            <p className="text-gray-600 text-[14px] leading-relaxed text-justify whitespace-pre-wrap">
                {article.summary}
            </p>
            
            <div className="text-xs text-gray-400 font-medium">
                引用元: {article.sourceName}
            </div>
            
            <div className="grid grid-cols-2 gap-2 mt-2">
                <button 
                    onClick={() => toggleRead(article)} 
                    className={`py-2.5 rounded-xl text-sm font-bold flex items-center justify-center gap-1.5 transition-colors ${article.isRead ? 'bg-gray-100 text-gray-500 hover:bg-gray-200' : 'bg-blue-50 text-blue-600 hover:bg-blue-100'}`}
                >
                    {article.isRead ? <CheckCircle size={18} /> : <Circle size={18} />}
                    {article.isRead ? '既読' : '既読にする'}
                </button>
                <button 
                    onClick={() => toggleSave(article)} 
                    className={`py-2.5 rounded-xl text-sm font-bold flex items-center justify-center gap-1.5 transition-colors ${article.isSaved ? 'bg-orange-100 text-orange-600 hover:bg-orange-200' : 'bg-orange-50/50 text-orange-500 hover:bg-orange-100'}`}
                >
                    <Bookmark size={18} fill={article.isSaved ? "currentColor" : "none"} />
                    {article.isSaved ? '保管済み' : '保管する'}
                </button>
            </div>
            <div className="grid grid-cols-2 gap-2">
                <a 
                    href={article.sourceUrl} 
                    target="_blank" 
                    rel="noreferrer" 
                    className="py-2.5 border border-gray-200 text-gray-600 rounded-xl text-sm font-bold flex items-center justify-center gap-1.5 hover:bg-gray-50 transition-colors"
                >
                    <Compass size={18} /> Safariで開く
                </a>
                <button 
                    onClick={() => handleCopyUrl(article.sourceUrl)} 
                    className="py-2.5 border border-gray-200 text-gray-600 rounded-xl text-sm font-bold flex items-center justify-center gap-1.5 hover:bg-gray-50 transition-colors"
                >
                    <Copy size={18} /> URLをコピー
                </button>
            </div>
        </div>
    );

    // Filters and Sorting
    const currentBatchId = settings.lastFetchedDate;
    // 今日のニュース: 現在のバッチに属し、未保管のもの（保管済みは保管タブで見る設計）
    let displayArticles = allArticles.filter(a => a.fetchBatchId === currentBatchId);
    
    if (tab === 'saved') {
        displayArticles = allArticles.filter(a => a.isSaved);
    }

    if (categoryFilter !== 'All') {
        displayArticles = displayArticles.filter(a => a.category?.includes(categoryFilter));
    }

    // 既読を下に、未読を上にソート
    displayArticles.sort((a, b) => {
        if (a.isRead === b.isRead) return b.createdAt - a.createdAt;
        return a.isRead ? 1 : -1;
    });

    const categories = ['All', '経済', '医療', 'IT'];

    return (
        <div className="min-h-screen bg-[#FFFBF5] text-slate-800 font-sans selection:bg-orange-200">
            {/* Header */}
            <header className="sticky top-0 z-30 bg-white/80 backdrop-blur-md border-b border-orange-100 px-4 py-3 flex items-center justify-between shadow-sm">
                <div className="flex items-center gap-2">
                    <MorningNewsIcon className="w-9 h-9" />
                    <h1 className="text-xl font-extrabold bg-gradient-to-r from-orange-400 to-rose-400 bg-clip-text text-transparent tracking-tight">
                        Morning News
                    </h1>
                </div>
                <button 
                    onClick={handleFetchNews} 
                    disabled={isFetching}
                    className="p-2 bg-orange-50 text-orange-500 rounded-full hover:bg-orange-100 transition-colors disabled:opacity-50"
                >
                    <RefreshCw size={20} className={isFetching ? "animate-spin" : ""} />
                </button>
            </header>

            {/* Main Content */}
            <main className="pb-24 pt-4 px-4 max-w-2xl mx-auto min-h-[80vh]">
                
                {/* ニュース表示タブ (Home & Saved) */}
                {(tab === 'home' || tab === 'saved') && (
                    <div className="flex flex-col gap-4">
                        <div className="flex items-center justify-between mb-2">
                            <h2 className="text-lg font-bold flex items-center gap-2 text-gray-800">
                                {tab === 'home' ? <Sun className="text-orange-400" size={20}/> : <Bookmark className="text-orange-400" size={20}/>}
                                {tab === 'home' ? (currentBatchId ? `${currentBatchId}のニュース` : '最新ニュース') : '保管したニュース'}
                            </h2>
                        </div>

                        {/* Category Filter */}
                        <div className="flex gap-2 overflow-x-auto pb-2 scrollbar-hide">
                            {categories.map(cat => (
                                <button
                                    key={cat}
                                    onClick={() => setCategoryFilter(cat)}
                                    className={`px-4 py-1.5 rounded-full text-sm font-bold whitespace-nowrap transition-all ${categoryFilter === cat ? 'bg-gray-800 text-white shadow-md' : 'bg-white text-gray-500 border border-gray-200 hover:bg-gray-50'}`}
                                >
                                    {cat === 'All' ? 'すべて' : cat}
                                </button>
                            ))}
                        </div>

                        {/* Loading State */}
                        {isFetching && (
                            <div className="flex flex-col items-center justify-center py-20 text-orange-400 gap-3">
                                <Loader2 size={40} className="animate-spin" />
                                <p className="font-bold text-sm">最新のニュースを集めています...</p>
                            </div>
                        )}

                        {/* Article List */}
                        {!isFetching && displayArticles.length > 0 ? (
                            <div className="flex flex-col gap-4">
                                {displayArticles.map(renderArticleCard)}
                            </div>
                        ) : (
                            !isFetching && (
                                <div className="text-center py-20 text-gray-400">
                                    <p className="font-medium text-lg">{tab === 'home' ? 'ニュースがありません' : '保管されたニュースはありません'}</p>
                                    {tab === 'home' && <p className="text-sm mt-2">右上のボタンから最新情報を取得してください</p>}
                                </div>
                            )
                        )}
                    </div>
                )}

                {/* 設定タブ */}
                {tab === 'settings' && (
                    <div className="animate-in fade-in slide-in-from-bottom-4 duration-300">
                        <h2 className="text-xl font-extrabold mb-6 flex items-center gap-2 text-gray-800">
                            <Settings className="text-gray-500" /> アプリ設定
                        </h2>
                        
                        <div className="bg-white p-6 rounded-3xl shadow-sm border border-orange-100 flex flex-col gap-6">
                            <div>
                                <label className="block text-sm font-bold text-gray-700 mb-3 flex items-center gap-2">
                                    <Sun size={18} className="text-orange-500" />
                                    毎朝のニュース更新時間
                                </label>
                                <input 
                                    type="time" 
                                    value={settings.time || "07:00"} 
                                    onChange={updateTimeSetting}
                                    className="w-full border-2 border-orange-100 focus:border-orange-400 outline-none rounded-2xl p-4 text-xl font-bold text-center text-gray-800 bg-orange-50/30 transition-colors"
                                />
                                <p className="text-xs text-gray-500 mt-4 leading-relaxed font-medium">
                                    指定した時間以降にアプリを開くと、経済・医療・ITの最新ニュースを自動でAIが取得・要約します。
                                </p>
                            </div>
                            
                            <hr className="border-gray-100" />
                            
                            <div>
                                <h3 className="text-sm font-bold text-gray-700 mb-2">データ管理</h3>
                                <p className="text-xs text-gray-500 mb-4 font-medium leading-relaxed">
                                    既読にした記事のタイトルは、新しいニュースを生成する際の除外リストとして保存されています。（現在 {settings.readHistory.length} 件）
                                </p>
                            </div>
                        </div>
                    </div>
                )}
            </main>

            {/* Bottom Navigation */}
            <nav className="fixed bottom-0 w-full bg-white/90 backdrop-blur-md border-t border-gray-100 flex justify-around py-3 pb-safe z-40 shadow-[0_-4px_20px_rgba(0,0,0,0.05)]">
                <NavItem icon={<Newspaper />} label="ホーム" active={tab === 'home'} onClick={() => {setTab('home'); setCategoryFilter('All');}} />
                <NavItem icon={<Bookmark />} label="保管済み" active={tab === 'saved'} onClick={() => {setTab('saved'); setCategoryFilter('All');}} />
                <NavItem icon={<Settings />} label="設定" active={tab === 'settings'} onClick={() => setTab('settings')} />
            </nav>

            {/* Toast Notification */}
            {toast && (
                <div className="fixed bottom-24 left-1/2 -translate-x-1/2 bg-gray-800/95 backdrop-blur text-white px-5 py-3 rounded-2xl shadow-xl z-50 text-sm font-bold whitespace-nowrap animate-in fade-in slide-in-from-bottom-4 flex items-center gap-2">
                    <CheckCircle size={16} className="text-green-400" />
                    {toast}
                </div>
            )}
        </div>
    );
}
